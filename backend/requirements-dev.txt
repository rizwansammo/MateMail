# Test-only. tests/test_deployment_workflow.py parses .github/workflows/*.yml
# and deploy/docker-compose.yml to assert the deployment invariants — that the
# deploy is manual-only, that no host-global docker prune runs on a shared box,
# that no `:latest` tag is published. Nothing in the application imports yaml,
# so this stays out of requirements.txt.
#
# It was undeclared until now and CI caught it: the module failed to import, so
# its 12 tests were silently never discovered and the suite reported 495
# instead of 506. A local environment that happened to have PyYAML installed
# directly hid this.
#
# The monitoring, NE6 and NE7 regression files parse the same way and import it
# too. They reached for `pytest.importorskip("yaml")` instead, which degrades to
# zero tests rather than an error — the softer form of the same bug, and the
# reason those suites were quietly absent from CI. Declared here, so they now
# import it plainly.
PyYAML==6.0.2

# Test-only, and declared for the same reason. tests/test_refresh_cookie.py
# decodes an access token to assert the tenant_id claim survives a refresh.
# PyJWT arrives anyway as a dependency of djangorestframework-simplejwt, so CI
# does not currently fail without this line — but a direct import that relies
# on someone else's transitive dependency breaks silently the day that library
# changes its JWT backend. Declare what you import.
PyJWT==2.13.0

# Test-only, and NOT in requirements.txt on purpose.
#
# The Native Engine API is a separate component with its own image and its own
# pinned dependency set (deploy/native-engine/images/api/requirements.txt). Its
# code must still be TESTED, and the repository has one test suite that CI runs,
# so the two libraries its tests import are declared here.
#
# They are deliberately absent from requirements.txt: the Django application
# does not import either, must never connect to the engine database, and must
# never hash a mailbox password. A production dependency would blur exactly the
# boundary NE0.2 exists to draw.
#
# Versions match the engine image exactly. Testing against a different bcrypt
# than the one that runs in production would be testing something else.
bcrypt==4.2.1
psycopg[binary]==3.2.3

# Test-only, NE3. tests/test_native_engine_mail_flow.py signs a message with the
# private key the engine actually stored and verifies it against the public
# material the Native API returns — the invariant that decides whether mail this
# engine sends is accepted by the Internet.
#
# It is the counterpart to the manual acceptance gate: a message produced by the
# real Postfix -> Rspamd -> LMTP flow was cryptographically verified with this
# same library against the API's public key, before and after a selector
# rotation. This test keeps that property from regressing without needing the
# whole stack running.
#
# Absent from requirements.txt deliberately: nothing in the Django application
# signs or verifies DKIM. Rspamd signs, and it has its own implementation.
dkimpy==1.1.8
