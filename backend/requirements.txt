# HTML sanitisation for PostBox (P11). Email HTML is the most hostile input
# MateMail renders, and `nh3` is a binding to Ammonia — a Rust allow-list
# sanitiser that parses with the same html5ever engine browsers use. A
# regex-based or string-replacement "sanitiser" is not a sanitiser: the whole
# attack surface is the gap between what the filter parses and what the browser
# parses, and only a real HTML parser closes it.
nh3==0.2.20

# Safe XML parsing for the Autodiscover endpoint. Outlook POSTs XML from
# the open internet, and `xml.etree` resolves external entities on some
# configurations — which turns a mail-client setup request into a file-read
# or an outbound fetch of the attacker's choosing. defusedxml is the same
# API with those features removed rather than a parser we have to remember
# to configure correctly.
defusedxml==0.7.1

google-auth==2.58.1
